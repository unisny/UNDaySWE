Your first fetch · JavaScript 
// Live earthquakes from the last 24 hours. No key needed.
const response = await fetch("https://earthquake.usgs.gov/earthquakes/feed/v1.0/summary/all_day.geojson");
const quakes = (await response.json()).features;
console.log("Earthquakes in the last 24 hours: " + quakes.length);

const biggest = quakes.reduce((a, b) => (b.properties.mag > a.properties.mag ? b : a));
console.log("Biggest: M" + biggest.properties.mag + ", " + biggest.properties.place);
